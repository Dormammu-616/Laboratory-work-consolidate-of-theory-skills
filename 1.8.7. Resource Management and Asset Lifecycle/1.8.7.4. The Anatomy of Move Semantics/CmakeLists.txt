cmake_minimum_required(VERSION 3.15)

# Название проекта и язык
project(MoveSemantics VERSION 1.0 LANGUAGES CXX)

# Задаем стандарт
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # Отключаем специфичные расширения компиляторов

# Добавляем исполняемый файл (наша точка сборки)
add_executable(MoveSemanticsApp
	src/main.cpp
	
	# lab 1 - Safe Resource Absorption
	include/lab1/CryptoBuffer.hpp
	include/lab1/GatewayRouter.hpp
	src/lab1/CryptoBuffer.cpp
	src/lab1/GatewayRouter.cpp

	# lab 2 - Perfect Forwarding
	include/lab2/SecurityEvent.hpp
	include/lab2/EventDispatcher.hpp
	src/lab2/SecurityEvent.cpp

	# lab 3 - Architecture Zero-Overhead
	include/lab3/LogAggregator.hpp
	src/lab3/LogAggregator.cpp
)

# Указываем, где искать заголовочные файлы
target_include_directories(MoveSemanticsApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Включаем максимальный уровень предупреждений (аналог /W4 и /WX)
if(MSVC)
    target_compile_options(MoveSemanticsApp PRIVATE /W4 /WX /analyze)
else()
    target_compile_options(MoveSemanticsApp PRIVATE -Wall -Wextra -Werror)
endif()
