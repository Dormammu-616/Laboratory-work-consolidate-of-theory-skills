cmake_minimum_required(VERSION 3.15)

# Название проекта и язык
project(OwnershipAndBorrowing VERSION 1.0 LANGUAGES CXX)

# Задаем стандарт C++20
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # Отключаем специфичные расширения компиляторов

# Добавляем исполняемый файл (наша точка сборки)
add_executable(OwnershipAndBorrowingApp
	src/main.cpp
	
	# lab 1 - Strict Contracts: Observers and Mutators
	include/lab1/Transaction.hpp
	include/lab1/FraudFilter.hpp
	include/lab1/FeeApplier.hpp
	include/lab1/TransactionProcessor.hpp

	src/lab1/Transaction.cpp
	src/lab1/FraudFilter.cpp
	src/lab1/FeeApplier.cpp
	src/lab1/TransactionProcessor.cpp


	# lab 2 - Zero-Overhead Borrowing
	include/lab2/Packet.hpp
	include/lab2/SignatureScanner.hpp
	include/lab2/PacketAnalyzer.hpp

	src/lab2/Packet.cpp
	src/lab2/SignatureScanner.cpp
	src/lab2/PacketAnalyzer.cpp
	

	# lab 3 - Safe monitoring
	include/lab3/ClientSession.hpp
	include/lab3/SessionManager.hpp
	include/lab3/SessionObserver.hpp

	src/lab3/ClientSession.cpp
	src/lab3/SessionManager.cpp
	src/lab3/SessionObserver.cpp
)

# Указываем, где искать заголовочные файлы
target_include_directories(OwnershipAndBorrowingApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Включаем максимальный уровень предупреждений (аналог /W4 и /WX)
if(MSVC)
    target_compile_options(OwnershipAndBorrowingApp PRIVATE /W4 /WX)
else()
    target_compile_options(OwnershipAndBorrowingApp PRIVATE -Wall -Wextra -Werror)
endif()
