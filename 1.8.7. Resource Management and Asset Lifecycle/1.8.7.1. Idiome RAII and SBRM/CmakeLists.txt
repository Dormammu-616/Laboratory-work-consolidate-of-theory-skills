cmake_minimum_required(VERSION 3.15)

# Название проекта и язык
project(IdiomeRAIIandSBRM VERSION 1.0 LANGUAGES CXX)

# Задаем стандарт C++20
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # Отключаем специфичные расширения компиляторов

# Добавляем разрешение использование std::fopen
add_compile_definitions(_CRT_SECURE_NO_WARNINGS)

# Добавляем исполняемый файл (наша точка сборки)
add_executable(IdiomeRAIIandSBRMApp
	src/main.cpp

	# lab 1 - Encapsulating a Legacy C-API
	include/lab1/FileDeleter.hpp
	include/lab1/AuditFileHandle.hpp
	include/lab1/AuditLogger.hpp
	src/lab1/AuditLogger.cpp
	
	# lab 2 - Transactional Construction
	include/lab2/MockFileHandle.hpp
	include/lab2/MockSocketHandle.hpp
	include/lab2/DatabaseNode.hpp
	
	# lab 3 - 
	
)

# Указываем, где искать заголовочные файлы
target_include_directories(IdiomeRAIIandSBRMApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Включаем максимальный уровень предупреждений (аналог /W4 и /WX)
if(MSVC)
    target_compile_options(IdiomeRAIIandSBRMApp PRIVATE /W4 /WX)
else()
    target_compile_options(IdiomeRAIIandSBRMApp PRIVATE -Wall -Wextra -Werror)
endif()
