cmake_minimum_required(VERSION 3.15)

# Название проекта и язык
project(TypeErasure VERSION 1.0 LANGUAGES CXX)

# Задаем стандарт C++17
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # Отключаем специфичные расширения компиляторов

# Добавляем исполняемый файл (наша точка сборки)
add_executable(TypeErasureApp
	src/main.cpp
	
	# lab 1 - Basic Concept/Model Idiom
	src/lab1/RenderPipeline.cpp
	
	# lab 2 - Type Erasure-Value Semantics
	src/lab2/SimulationGrid.cpp
)

# Указываем, где искать заголовочные файлы
target_include_directories(TypeErasureApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Включаем максимальный уровень предупреждений (аналог /W4 и /WX)
if(MSVC)
    target_compile_options(TypeErasureApp PRIVATE /W4 /WX)
else()
    target_compile_options(TypeErasureApp PRIVATE -Wall -Wextra -Werror)
endif()
