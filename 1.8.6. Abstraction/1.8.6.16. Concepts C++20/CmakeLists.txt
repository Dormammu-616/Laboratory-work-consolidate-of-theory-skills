cmake_minimum_required(VERSION 3.15)

# Название проекта и язык
project(Concept VERSION 1.0 LANGUAGES CXX)

# Задаем стандарт C++17
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF) # Отключаем специфичные расширения компиляторов

# Добавляем исполняемый файл (наша точка сборки)
add_executable(ConceptApp
	src/main.cpp

	# lab 1 - Zero-Overhead Contracts
	include/lab1/SerializableConcept.hpp
	include/lab1/UserRecord.hpp
	include/lab1/TransactionRecord.hpp
	include/lab1/SystemState.hpp
	include/lab1/DataExporter.hpp
	
	# lab 2 - Intelligent routing and standard concepts
	include/lab2/AlgorithmRouter.hpp
	include/lab2/ExecutionEngine.hpp
	
	# lab 3 - Compound & Nested Requirements
	include/lab3/BusinessConcepts.hpp
	include/lab3/DataModels.hpp
	include/lab3/DataProcessor.hpp
)

# Указываем, где искать заголовочные файлы
target_include_directories(ConceptApp PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)

# Включаем максимальный уровень предупреждений (аналог /W4 и /WX)
if(MSVC)
    target_compile_options(ConceptApp PRIVATE /W4 /WX)
else()
    target_compile_options(ConceptApp PRIVATE -Wall -Wextra -Werror)
endif()
